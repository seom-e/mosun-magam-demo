function readFiles(files){
  var left = files.length;
  if (!left) return;
  Array.prototype.forEach.call(files, function(f){
    var rd = new FileReader();
    var isX = /\.(xls|xlsx)$/i.test(f.name);
    rd.onload = function(e){
      try {
        var what = null;
        if (isX) {
          if (typeof XLSX === 'undefined') throw new Error('엑셀 읽는 부품을 못 불러왔습니다 (인터넷 확인)');
          what = takeExcel(XLSX.read(e.target.result, {type:'binary'}), f.name);
        } else {
          var txt = e.target.result;
          var m = String(txt).match(/^\s*window\.\w+\s*=\s*([\s\S]*?);?\s*$/);   /* board.js / mail.js */
          what = takeJson(JSON.parse(m ? m[1] : txt), f.name);
        }
        say(what ? ('읽음  ' + f.name + '  →  ' + what)
                 : ('모르는 형식  ' + f.name + ' — 건너뜁니다'));
      } catch (err) {
        say('못 읽음  ' + f.name + ' — ' + err.message);
      }
      if (--left === 0) render();
    };
    rd.onerror = function(){ say('못 읽음  ' + f.name); if (--left === 0) render(); };
    if (isX) rd.readAsBinaryString(f); else rd.readAsText(f, 'utf-8');
  });
}

      });

OK, WORK, ERR, NONE, MINE = "승인", "진행중", "오류", "미신고", "타서비스"
MARK = {OK: "○", WORK: "△", ERR: "✗", NONE: "✗", MINE: "–"}


def _mrn(cell: str) -> str | None:
    """체크리스트 MRN 칸에서 진짜 MRN 만 꺼낸다. `BSS1 IN` 같은 메모는 아직 미발급."""
    s = (cell or "").strip().upper()
    if len(s) >= 10 and s[:2].isdigit() and s[-1] in "IE":
        return s
    return None


def find_row(cell: str, name: str, vygs: list[str], rows: list[dict]) -> tuple[dict | None, str]:
    """PLISM 결과에서 이 모선의 줄을 찾는다.

    MRN 으로 먼저 찾고, 없으면 선명+항차로 찾는다.
    체크리스트 MRN 칸이 비어 있어도(아직 안 옮겨 적었어도) 신고는 되어 있을 수 있기 때문이다.
    """
    mrn = _mrn(cell)
    if mrn:
        for r in rows:
            if r["mrn"] == mrn:
                return r, ""
    hit = [r for r in rows
           if (r["vsl"] or "").strip().upper() == (name or "").strip().upper()
           and any(v and (r["vyg"] or "").strip().upper() == v.strip().upper() for v in vygs)]
    if len(hit) == 1:
        return hit[0], ("MRN 다름" if mrn else "체크리스트에 MRN 미기재")
    return None, ""


def judge_mf(cell: str, row: dict | None, how: str = "") -> tuple[str, str]:
    """적하목록(수입·수출) 한 칸을 판정한다. 돌려주는 값은 (판정, 설명)."""
    if row is None:
        memo = (cell or "").strip()
        if memo and not _mrn(memo):
            return MINE, memo            # `BSS1 IN` — 그 방향은 타 서비스 소속이다
        return NONE, "PLISM 에 없음"
    tail = f" ({how})" if how else ""
    if row.get("err"):
        return ERR, f"오류 B/L {row['err']}건{tail}"
    status = (row.get("st") or "").strip()
    if status == "승인":
        return OK, "승인" + tail
    if status:
        return WORK, status + tail       # 선별마감 · 1차 접수 · 2차 접수 …
    return WORK, "상태 없음" + tail


def judge_dg(rows: list[dict]) -> tuple[str, str]:
    """위험물 한 칸을 판정한다.

    한 모선에 응답이 여러 줄 쌓인다 — 문서 종류별로 가장 최근 것만 본다.
    그러지 않으면 이미 고친 어제 오류가 오늘도 계속 뜬다.
    """
    if not rows:
        return NONE, "신고 없음"

    latest: dict[str, dict] = {}
    for r in rows:
        for res in r.get("res") or []:
            doc, at = res.get("doc"), str(res.get("at") or "")
            if doc and (doc not in latest or at > str(latest[doc].get("at") or "")):
                latest[doc] = res

    if not latest:
        return WORK, "전송됨 · 응답 대기"

    bad = [f"{d} {r.get('code') or ''}".strip()
           for d, r in latest.items() if (r.get("stat") or "") != "(정상처리)"]
    if bad:
        return ERR, " / ".join(bad)

    if any(r.get("wEq") == "N" or r.get("cEq") == "N" for r in rows):
        return WORK, "정상처리 (중량·컨 수량 불일치)"

    return OK, f"정상처리 {len(latest)}종"

## 체크리스트와 어떻게 맞물리는가

조회에 필요한 열쇠 세 가지가 **이미 전부 체크리스트 쪽에 있다.**

| 조회 열쇠 | 지금 어디 있나 |
|---|---|
| 수입 MRN | 체크리스트 `AG` 열 = `.체크리스트이력\*.json` 의 `imp_cell` |
| 수출 MRN | 체크리스트 `AH` 열 = `exp_cell` |
| 호출부호 | 체크리스트 `Z` 열, `마감관리v4.34.html` 의 `CALLSIGN_MAP`, `선박 파티큘러-콜사인정보.xls` |
| 해수청 번호 | 체크리스트 `J`·`AD` 열 = `seq` → DG 조회의 `arv_ser_no` |

실제로 맞는지 대조해 확인했다 (2026-09-16 기준):

| 체크리스트 | PLISM 조회 결과 |
|---|---|
| `PCBJ 2619W` 수출MRN `26SNKO3459E`, 해수청 `011` | 수출목록 `26SNKO3459E` / `PACIFIC BEIJING 2619W` / `승인` · DG `9V8000 2026-011` `(정상처리)` |
| `SWSH 2634E` 수입MRN `26SNKO3455I` | 수입목록 `26SNKO3455I` / `SAWASDEE SHANGHAI 2634E` / `선별마감` |

즉 **체크리스트를 만들면 그것이 그대로 조회 열쇠 꾸러미가 된다.**
따로 입력할 것이 없다.

## 하루 한 바퀴

```
스케줄 확인  →  체크리스트 작성  →  (MRN·호출부호 확보)  →  신고  →  신고 확인
   vessel-checklist 스킬          체크리스트가 열쇠        PLISM      ← 이 문서
```

마지막 '신고 확인' 칸이 지금까지 사람 눈으로 하던 자리다.
`마감관리v4.34.html` 에 이미 `plism_imp_status` / `plism_exp_status`
(`제출전`/`1차오류`/`승인`) 칸이 있는데 지금은 손으로 고른다 — 여기에 자동 수집 결과를 넣는다.

## 남은 확인거리

- `status_cusagd` 가 정말 하선신고 상태인지 (하선신고 전용 화면이 따로 있는지)
- `status_cus` 가 가질 수 있는 값 전체 (`선별마감` `승인` 외에 `1차 접수` `2차 접수` `제출전` 등)
- 오류가 실제로 난 건을 하나 봐야 오류 표시가 어떻게 오는지 확정된다

## 판정 규칙 — 체크리스트가 기준이다

PLISM만 보면 **미신고를 찾을 수 없다.** 신고한 것만 나오기 때문이다.
그래서 **체크리스트의 모선 목록을 모수(母數)로 두고**, 거기에 PLISM 조회 결과를 붙인다.
붙지 않고 남은 줄이 곧 미신고다. (5/28 메일의 `조회 미매칭` 이 이 자리였다)

각 모선의 수입/수출/위험물 세 칸을 이렇게 판정한다.

| 판정 | 조건 | 표시 |
|---|---|---|
| **미신고** | 체크리스트에 MRN이 적혀 있는데 PLISM 목록에 그 MRN이 없음 | 🔴 조치 |
| **대기** | MRN 칸이 아직 메모(`BSS1 IN`)이거나 비어 있음 — 아직 발급 전 | ⚪ 대상 아님 |
| **오류** | `err_bl_cnt > 0` · 상세의 `cust_err_mbl_cnt`·`mismatch_cnt > 0` | 🔴 조치 |
| **선별마감** | `status_cus = 선별마감` | 🟡 진행 |
| **진행중** | `status_cus` 가 접수 단계 값 | 🟡 진행 |
| **승인** | `status_cus = 승인` | 🟢 완료 |
| **건수 불일치** | `tot_cusagd_mbl ≠ tot_cms_imp_mbl` | 🟡 확인 |

위험물(622)은 호출부호+해수청번호로 조회해:

| 판정 | 조건 |
|---|---|
| **미신고** | 체크리스트에 DG 있는 모선인데 `DgLineMngList` 결과가 비어 있음 |
| **오류** | `DgCombineResList` 의 `freetxt2` 가 `(정상처리)` 가 아님 |
| **승인** | `freetxt2 = (정상처리)` |
| **수량 불일치** | `weight_equal_yn` 또는 `cntr_equal_yn` 가 `N` |

미신고·오류만 추려 알림 메일로 보내고, 전체 판정은 화면에 표로 남긴다.

## 오류 판정 — 확정 (2026-09-16)

`selectSubmitInfo.do` 응답의 `resultSubmitHis[]` 가 **제출 차수별 이력**이다.
화면의 '오류통보 수신일시' 가 여기 `error_date` 다.

| 칸 | 오늘 값 | 뜻 |
|---|---|---|
| `status_cus` | `S1` / `S2` | **제출 차수** — S1=1차, S2=2차 |
| `send_date` | `20XX-XXX-XXXX` | 제출(송신) 일시 |
| `res_date` | `20XX-XXX-XXXX` | 접수 통보 일시 |
| `error_date` | `None` | **오류통보 수신일시 — 값이 있으면 그 차수는 오류** |
| `end_date` | `20XX-XXX-XXXX` | 마감(완료) 일시 |

오늘은 오류 건이 없어 전부 `None` 인 것을 확인했다. 오류가 나면 여기에 날짜·시각이 찍힌다.

이것으로 5/28 메일의 상태 표현이 어디서 나왔는지가 풀린다:

| 메일 표현 | 실제 근거 |
|---|---|
| `1차 접수` | `S1` 이 있고 `error_date` 없음 |
| `1차 오류` | `S1` 의 `error_date` 에 값 있음 |
| `2차 접수` | `S2` 가 있음 |
| `선별마감` | 목록의 `status_cus = 선별마감` |
| `제출전` | 이력이 비어 있음 |
| `조회 미매칭` | 체크리스트에 있는 MRN이 PLISM 목록에 없음 |

→ 오류 판정은 **목록의 `err_bl_cnt` 로 1차 감지, 상세의 `error_date` 로 확정**한다.
   오류가 언제 통보됐는지까지 알 수 있으므로 "몇 시간째 방치된 오류" 도 잡아낼 수 있다.

## 622 위험물 판정 — 확정

화면의 **응답상태 / 응답내용** 두 칸만 보면 된다. `DgCombineResList.do` 응답에서:

| 화면 | 칸 | 예시 |
|---|---|---|
| 문서 | `docnm` | `위험물일람표` / `위험물반입신고서` |
| **응답상태** | `freetxt2` | `(정상처리)` ← 여기가 승인/오류 판정 |
| **응답내용** | `restxt` | `... 위험물 일람표신고 자료가입력 되었음.` (오류면 사유가 여기 들어온다) |
| 응답코드 | `errmsg` | `MSG-18001` |
| 응답일시 | `genrestm` | `2026-09-15 15:37:24` |

주의: **한 모선에 여러 줄이 온다.** 문서 종류(일람표·반입신고서)별로, 그리고 입력·수정할
때마다 한 줄씩 쌓인다 (오늘 PCBJ = 3줄). 그러므로 판정은
**문서 종류별로 `genrestm` 이 가장 최근인 줄 하나씩**을 보고 한다.
그 최신 줄의 `freetxt2` 가 `(정상처리)` 가 아니면 오류, 내용은 `restxt` 를 그대로 보여준다.

## 신고 마감시한 규칙

미결 판정의 기준. **수입과 수출의 근거리 지역 목록이 다르다** — 이걸 헷갈리면 안 된다.

### 수입 (적하목록)

| 조건 | 마감 |
|---|---|
| 전출항지가 **근거리** | 입항 전까지 |
| 그 외 (원거리) | **입항 24시간 전** |

근거리(수입): `KR CN JP RU TW HK`

### 수출 (적하목록)

| 조건 | 마감 |
|---|---|
| 차항지가 **근거리** | 선박 적재 전 · **출항 30분 전 최종마감** |
| 그 외 (원거리) | **적재 24시간 전** |

근거리(수출): `CN TW HK JP RU PH VN KH(캄보디아) TH ID(인도네시아) MY SG`

→ 수출 근거리는 동남아까지 포함해 훨씬 넓다. 수입 근거리에만 `KR` 이 있다.

### 서비스별 예외

| 서비스 | 수입 | 수출 |
|---|---|---|
| **KI1** | 전출항 `SUB-HKG-광양1ST`. 보통은 HKG(근거리)에서 오지만 **HKG 를 자주 스킵**한다. IDSUB(인도네시아)에서 바로 오면 **원거리 → 24시간 전** | 근거리 → **출항 30분 전** |
| **SIS2** | `MYPKL/부산`, 원거리. 참조표 유의사항에도 `(원거리)/입출항 24H전 신고` 라고 적혀 있다 | **INNSA · INMUN(인도)** 행이 있어 **24시간 전** |

한 서비스 안에서도 수입은 원거리, 수출은 근거리일 수 있다. 반드시 방향별로 따로 판정한다.

### 전출항지·차항지는 어디서 오나

체크리스트 참조표 `AK4:AU25` 의 `AO`(전출항) · `AP`(차항지) 에 서비스별로 적혀 있다.

| 서비스 | 전출항(AO) | 차항지(AP) |
|---|---|---|
| KI1 | `SUB-HKG-광양1ST` | 울산 |
| SIS2 | `MYPKL/부산` | 광양 |
| NSB | `MNN-광양1ST` | 부산 |
| KXS1 | `TSN/부산/울산` | 광양 |
| FEM | `SHA/부산/울산` | 광양 |

**한계**: 이건 서비스의 평상시 경로다. 실제 그 항차의 전출항지는 스킵 여부에 따라 달라진다
(KI1 이 대표적이고, KBX 도 참조표에 `스킵多` 라고 적혀 있다).

   왜 그 시각이 나왔는지 보이게 한다.

## 스킵 메일 — 전출항지를 여기서 얻는다

`COASTAL SCHEDULE` 메일 본문이 **고정 폭 표**라 그대로 읽힌다.

- 발신: `발신전용4 <demo36@example.com>` (Sinokor Line Operation Team)
- 제목 예: `// 2510N HKG SKIP NOTICE // [KI1] M/V PORT KLANG VOYAGER 2509/2510 COASTAL SCHEDULE(10/22)`
- 제목에서 서비스(`[KI1]`)와 선박명을 얻는다

본문:

```
SEQ VYG(B)  From    E.T.A             DAY  E.T.B             DAY  E.T.D             DAY  DLY(A)
08  2509N  IDSUB  2025-10-22 06:00  Wed  2025-10-22 08:00  Wed  2025-10-23 08:00  Thu  434:00
09  2509N  HKHKG  SKIP
01  2510S  KRKAN  2025-10-30 19:00  Thu  2025-10-30 20:30  Thu  2025-10-31 04:30  Fri  340:00
```

**전출항지 구하는 법**: `KRKAN`(광양) 줄을 찾고, 거기서 위로 올라가며
`SKIP` 이 아닌 첫 항구를 취한다. 위 예에서는 `HKHKG` 가 SKIP 이므로 → **`IDSUB`**.
`ID` 는 수입 근거리 목록에 없으므로 → **원거리 → 입항 24시간 전**.

이 표에는 전체 기항 순서가 다 들어 있으므로 **KI1 뿐 아니라 모든 서비스의 전출항지**를
같은 방법으로 얻을 수 있다. 차항지(광양 다음 줄)도 같이 나온다.

주의:
- 한 메일에 여러 항차(`2509N` `2510S` `2510N`)가 섞여 있다. 우리 항차와 맞는 줄을 골라야 한다
- `*2509S KRUSN DELAYED DUE TO BERTH CONGESTION` 처럼 표 아래에 지연 사유가 붙는다
- 이 메일을 **놓치는 것**이 지금의 실제 문제다. 자동으로 읽어 체크리스트에 반영하면 해결된다
